    {
        try {
            $this->userService->registration($request);
            return $this->successResponse();
        } catch (ValidationException $ve) {
            return $this->errorResponse(
                $ve->validator->errors()->toArray(),
                'Validation error',
                422
            );
        } catch (Exception $e) {
            return $this->exceptionResponse($e);
        }
    }
